cmake_minimum_required(VERSION 3.24)
project(FDTDStudio VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Core Gui Widgets OpenGLWidgets Test)
find_package(VTK 9.2 REQUIRED COMPONENTS CommonCore CommonDataModel CommonExecutionModel
    FiltersGeneral GUISupportQt InteractionStyle IOImage IOXML RenderingCore RenderingOpenGL2 RenderingAnnotation)
add_library(studio_core STATIC src/Project.cpp src/Project.h src/Canvas.cpp src/Canvas.h
    src/Plot.cpp src/Plot.h src/Dialogs.cpp src/Dialogs.h src/FieldView.cpp src/FieldView.h
    src/MainWindow.cpp src/MainWindow.h src/Inspection.cpp src/Inspection.h src/Ribbon.cpp src/Ribbon.h
    src/Radiation.cpp src/Radiation.h src/SnapshotPlayer.cpp src/SnapshotPlayer.h)
target_include_directories(studio_core PUBLIC src)
target_link_libraries(studio_core PUBLIC Qt6::Widgets Qt6::OpenGLWidgets ${VTK_LIBRARIES})
get_filename_component(FDTD_REPOSITORY_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
target_compile_definitions(studio_core PUBLIC FDTD_SOURCE_ROOT="${FDTD_REPOSITORY_ROOT}")
if(MSVC)
    target_compile_options(studio_core PRIVATE /W4 /utf-8)
else()
    target_compile_options(studio_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(fdtd-studio WIN32 src/main.cpp)
target_link_libraries(fdtd-studio PRIVATE studio_core)
vtk_module_autoinit(TARGETS studio_core fdtd-studio MODULES ${VTK_LIBRARIES})
include(CTest)
if(BUILD_TESTING)
    add_executable(studio_tests "${FDTD_REPOSITORY_ROOT}/tests/fdtd_2d/gui/test_studio.cpp")
    target_link_libraries(studio_tests PRIVATE studio_core Qt6::Test)
    vtk_module_autoinit(TARGETS studio_tests MODULES ${VTK_LIBRARIES})
    add_test(NAME studio_tests COMMAND studio_tests)
    set_tests_properties(studio_tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
endif()
install(TARGETS fdtd-studio RUNTIME DESTINATION bin)
